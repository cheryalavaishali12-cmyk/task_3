    Emp_ID INTEGER PRIMARY KEY,
    Emp_Name VARCHAR(50),
    Age INTEGER,
    Salary DECIMAL(10,2),
    Dept_ID INTEGER,
    FOREIGN KEY (Dept_ID) REFERENCES Department(Dept_ID)

-- ------------------------------------------------------------
-- Basic Queries
-- ------------------------------------------------------------

-- Select specific columns
SELECT Emp_id, Emp_name FROM Employee;

-- Filter with WHERE clause
SELECT * FROM Employee WHERE salary >= 40000;

-- Order results by salary descending
SELECT * FROM Employee ORDER BY Salary DESC;

-- Aggregations & Grouping
SELECT Dept_ID, AVG(Salary) AS Average_Salary FROM Employee GROUP BY Dept_ID;

SELECT MAX(Salary) AS Highest_Salary, MIN(Salary) AS Lowest_Salary FROM Employee;

-- ------------------------------------------------------------
-- Joins
-- ------------------------------------------------------------

-- Inner Join: Get employee details along with department name and location
SELECT 
    Employee.Emp_ID,
    Employee.Emp_Name,
    Department.Dept_Name,
    Department.Location
FROM Employee
INNER JOIN Department ON Employee.Dept_ID = Department.Dept_ID;

-- Left Join: Get employee names and department names
SELECT 
    Employee.Emp_Name,
    Department.Dept_Name
FROM Employee
LEFT JOIN Department ON Employee.Dept_ID = Department.Dept_ID;

-- Right Join: Get employee names and department names
SELECT 
    Employee.Emp_Name,
    Department.Dept_Name
FROM Employee
RIGHT JOIN Department ON Employee.Dept_ID = Department.Dept_ID;

-- ------------------------------------------------------------
-- Advanced Queries (Subqueries, Group By + Having, Views)
-- ------------------------------------------------------------

-- Subquery: Get employees belonging to departments in Hyderabad or Bangalore
SELECT * 
FROM Employee 
WHERE Dept_ID IN (
    SELECT Dept_ID 
    FROM Department 
    WHERE Location IN ('Hyderabad', 'Bangalore')
);

-- Group By, Where, and Having: Count employees per department with salary > 30000 where count > 1
SELECT 
    Dept_ID,
    COUNT(*) AS Employee_Count
FROM Employee
WHERE Salary > 30000
GROUP BY Dept_ID
HAVING COUNT(*) > 1;

-- View Creation & Querying
CREATE VIEW Employee_Details AS
SELECT 
    Employee.Emp_ID,
    Employee.Emp_Name,
    Employee.Salary,
    Department.Dept_Name,
    Department.Location
FROM Employee
INNER JOIN Department ON Employee.Dept_ID = Department.Dept_ID;

SELECT * FROM Employee_Details;

-- Department Analysis: Find department with the highest average salary
SELECT 
    Department.Dept_Name,
    AVG(Employee.Salary) AS Average_Salary
FROM Employee
INNER JOIN Department ON Employee.Dept_ID = Department.Dept_ID
GROUP BY Department.Dept_Name
ORDER BY Average_Salary DESC
LIMIT 1;
